<select id="confidence" onchange="updateChart()">

  <option value="20">Very Low</option>

  <option value="40">Low</option>

  <option value="60">Medium</option>

  <option value="80" selected>High</option>

  <option value="100">Very High</option>

  <circle r="100" stroke="#ccc" fill="none"/>

  <circle r="80" stroke="#eee" fill="none"/>

  <line x1="0" y1="0" x2="100" y2="0" stroke="#aaa"/>

  <text x="110" y="5" font-size="14">Confidence</text>

  <polygon id="shape" fill="rgba(0, 123, 255, 0.4)" stroke="#007bff"></polygon>

</svg>

 

<script>

function updateChart() {

  const value = parseInt(document.getElementById("confidence").value);

  const angle = 0; // Single variable = 0 degrees

  const x = value * Math.cos(angle * Math.PI / 180);

  const y = value * Math.sin(angle * Math.PI / 180);

  document.getElementById("shape").setAttribute("points", `0,0 ${x},${y}`);

}

 

// Initialize chart

updateChart();

</script>
